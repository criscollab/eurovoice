    48→/**
    49→ * AdminPanel
    50→ * Slide-out panel for managing stations (CRUD) and songs (CRUD + reorder).
    51→ * Updates the global store after mutations so the player reflects changes.
    52→ */
    53→export function AdminPanel() {
    54→  const { adminOpen, setAdminOpen, stations, setStations } = useRadioStore()
    55→  const { toast } = useToast()
    56→
    57→  const [editingStation, setEditingStation] = useState<StationWithSongs | null>(null)
    58→  const [loadingStation, setLoadingStation] = useState<StationWithSongs | null>(null)
    59→  const [loadedStations, setLoadedStations] = useState<StationWithSongs[]>([])
    60→
    61→  const reloadLoadedStations = useCallback(async () => {
    62→    const res = await fetch('/api/stations', { cache: 'no-store' })
    63→    const data = await res.json()
    64→    const list: StationWithCount[] = data.stations ?? []
    65→    setStations(list)
    66→    const withSongs = await Promise.all(
    67→      list.map((s) =>
    68→        fetch(`/api/stations/${s.id}`, { cache: 'no-store' })
    69→          .then((r) => r.json())
    70→          .then((d) => d?.station as StationWithSongs)
    71→          .catch(() => null)
    72→      )
    73→    )
    74→    const valid = withSongs.filter(Boolean) as StationWithSongs[]
    75→    // Preserve the order from the stations list
    76→    const ordered = list
    77→      .map((s) => valid.find((v) => v.id === s.id))
    78→      .filter(Boolean) as StationWithSongs[]
    79→    setLoadedStations(ordered)
    80→  }, [setStations])
    81→
    82→  // === Load stations (with songs) when the admin opens ===
    83→  useEffect(() => {
    84→    if (!adminOpen) return
    85→    let cancelled = false
    86→    Promise.resolve().then(() => {
    87→      if (cancelled) return
    88→      reloadLoadedStations().catch(() => {})
    89→    })
    90→    return () => {
    91→      cancelled = true
    92→    }
    93→  }, [adminOpen, reloadLoadedStations])
    94→
    95→  const refreshStations = async () => {
    96→    // Reload both the store (lightweight list) and the loadedStations (with songs)
    97→    await reloadLoadedStations()
    98→  }
    99→
   100→  // === Station mutations ===
   101→  const handleCreateStation = async (data: {
   102→    name: string
   103→    description?: string
   104→    language?: string
   105→    color?: string
   106→  }) => {
   107→    try {
   108→      const res = await fetch('/api/stations', {
   109→        method: 'POST',
   110→        headers: { 'Content-Type': 'application/json' },
   111→        body: JSON.stringify(data),
   112→      })
   113→      if (!res.ok) throw new Error('create failed')
   114→      await refreshStations()
   115→      toast({ title: 'Emisora creada', description: `"${data.name}" está lista.` })
   116→    } catch {
   117→      toast({ title: 'Error', description: 'No se pudo crear la emisora.', variant: 'destructive' })
   118→    }
   119→  }
   120→
   121→  const handleUpdateStation = async (id: string, data: Partial<Station>) => {
   122→    try {
   123→      const res = await fetch(`/api/stations/${id}`, {
   124→        method: 'PATCH',
   125→        headers: { 'Content-Type': 'application/json' },
   126→        body: JSON.stringify(data),
   127→      })
   128→      if (!res.ok) throw new Error('update failed')
   129→      const d = await res.json()
   130→      setLoadedStations((prev) =>
   131→        prev.map((s) => (s.id === id ? { ...s, ...d.station, songs: s.songs } : s))
   132→      )
   133→      await refreshStations()
   134→      toast({ title: 'Emisora actualizada' })
   135→    } catch {
   136→      toast({ title: 'Error', description: 'No se pudo actualizar la emisora.', variant: 'destructive' })
   137→    }
   138→  }
   139→
   140→  const handleDeleteStation = async (id: string, name: string) => {
   141→    if (!confirm(`¿Eliminar la emisora "${name}" y todas sus canciones?`)) return
   142→    try {
   143→      const res = await fetch(`/api/stations/${id}`, { method: 'DELETE' })
   144→      if (!res.ok) throw new Error('delete failed')
   145→      setLoadedStations((prev) => prev.filter((s) => s.id !== id))
   146→      await refreshStations()
   147→      toast({ title: 'Emisora eliminada' })
   148→    } catch {
   149→      toast({ title: 'Error', description: 'No se pudo eliminar la emisora.', variant: 'destructive' })
   150→    }
   151→  }
   152→
   153→  // === Song mutations ===
   154→  const handleAddSong = async (
   155→    stationId: string,
   156→    data: { title: string; artist?: string; audioUrl: string; duration?: number; coverUrl?: string }
   157→  ) => {
   158→    try {
   159→      const res = await fetch(`/api/stations/${stationId}/songs`, {
   160→        method: 'POST',
   161→        headers: { 'Content-Type': 'application/json' },
   162→        body: JSON.stringify(data),
   163→      })
   164→      if (!res.ok) throw new Error('add song failed')
   165→      const d = await res.json()
   166→      setLoadedStations((prev) =>
   167→        prev.map((s) =>
   168→          s.id === stationId ? { ...s, songs: [...s.songs, d.song] } : s
   169→        )
   170→      )
   171→      await refreshStations()
   172→      toast({ title: 'Canción agregada', description: d.song.title })
   173→    } catch {
   174→      toast({ title: 'Error', description: 'No se pudo agregar la canción.', variant: 'destructive' })
   175→    }
   176→  }
   177→
   178→  const handleUpdateSong = async (songId: string, stationId: string, data: Partial<Song>) => {
   179→    try {
   180→      const res = await fetch(`/api/songs/${songId}`, {
   181→        method: 'PATCH',
   182→        headers: { 'Content-Type': 'application/json' },
   183→        body: JSON.stringify(data),
   184→      })
   185→      if (!res.ok) throw new Error('update song failed')
   186→      const d = await res.json()
   187→      setLoadedStations((prev) =>
   188→        prev.map((s) =>
   189→          s.id === stationId
   190→            ? { ...s, songs: s.songs.map((song) => (song.id === songId ? d.song : song)) }
   191→            : s
   192→        )
   193→      )
   194→      toast({ title: 'Canción actualizada' })
   195→    } catch {
   196→      toast({ title: 'Error', description: 'No se pudo actualizar la canción.', variant: 'destructive' })
   197→    }
   198→  }
   199→
   200→  const handleDeleteSong = async (songId: string, stationId: string) => {
   201→    if (!confirm('¿Eliminar esta canción?')) return
   202→    try {
   203→      const res = await fetch(`/api/songs/${songId}`, { method: 'DELETE' })
   204→      if (!res.ok) throw new Error('delete song failed')
   205→      setLoadedStations((prev) =>
   206→        prev.map((s) =>
   207→          s.id === stationId ? { ...s, songs: s.songs.filter((song) => song.id !== songId) } : s
   208→        )
   209→      )
   210→      await refreshStations()
   211→      toast({ title: 'Canción eliminada' })
   212→    } catch {
   213→      toast({ title: 'Error', description: 'No se pudo eliminar la canción.', variant: 'destructive' })
   214→    }
   215→  }
   216→
   217→  const handleMoveSong = async (stationId: string, index: number, direction: -1 | 1) => {
   218→    const station = loadedStations.find((s) => s.id === stationId)
   219→    if (!station) return
   220→    const songs = [...station.songs].sort((a, b) => a.order - b.order)
   221→    const newIndex = index + direction
   222→    if (newIndex < 0 || newIndex >= songs.length) return
   223→    // Swap
   224→    ;[songs[index], songs[newIndex]] = [songs[newIndex], songs[index]]
   225→    // Reassign orders
   226→    songs.forEach((s, i) => (s.order = i))
   227→    // Optimistic update
   228→    setLoadedStations((prev) =>
   229→      prev.map((s) => (s.id === stationId ? { ...s, songs } : s))
   230→    )
   231→    // Persist
   232→    try {
   233→      await fetch(`/api/stations/${stationId}/reorder`, {
   234→        method: 'POST',
   235→        headers: { 'Content-Type': 'application/json' },
   236→        body: JSON.stringify({ orderedSongIds: songs.map((s) => s.id) }),
   237→      })
   238→      toast({ title: 'Orden actualizado' })
   239→    } catch {
   240→      toast({ title: 'Error', description: 'No se pudo guardar el orden.', variant: 'destructive' })
   241→    }
   242→  }
   243→
   244→  // === Station reordering (move station up/down) ===
   245→  const handleMoveStation = async (index: number, direction: -1 | 1) => {
   246→    const stations = [...loadedStations].sort((a, b) => a.order - b.order)
   247→    const newIndex = index + direction
   248→    if (newIndex < 0 || newIndex >= stations.length) return
   249→    // Swap
   250→    ;[stations[index], stations[newIndex]] = [stations[newIndex], stations[index]]
   251→    // Reassign orders
   252→    stations.forEach((s, i) => (s.order = i))
   253→    // Optimistic update
   254→    setLoadedStations([...stations])
   255→    try {
   256→      await fetch('/api/stations/reorder', {
   257→        method: 'POST',
   258→        headers: { 'Content-Type': 'application/json' },
   259→        body: JSON.stringify({ orderedStationIds: stations.map((s) => s.id) }),
   260→      })
   261→      await refreshStations()
   262→      toast({ title: 'Orden de emisoras actualizado' })
   263→    } catch {
   264→      toast({ title: 'Error', description: 'No se pudo guardar el orden.', variant: 'destructive' })
   265→    }
   266→  }
   267→
   268→  return (
   269→    <Sheet open={adminOpen} onOpenChange={setAdminOpen}>
   270→      <SheetContent
   271→        side="right"
   272→        className="w-full overflow-hidden border-l-border/60 bg-background p-0 sm:max-w-2xl"
   273→      >
   274→        <SheetHeader className="border-b border-border/60 px-6 py-4">
   275→          <SheetTitle className="flex items-center gap-2 text-xl">
   276→            <Music className="h-5 w-5 text-primary" />
   277→            Panel de administración
   278→          </SheetTitle>
   279→          <SheetDescription>
   280→            Gestiona emisoras y canciones. Los cambios se aplican en vivo.
   281→          </SheetDescription>
   282→        </SheetHeader>
   283→
   284→        <ScrollArea className="h-[calc(100vh-100px)]">
   285→          <div className="p-6">
   286→            <Tabs defaultValue="stations">
   287→              <TabsList className="grid w-full grid-cols-2 mb-6">
   288→                <TabsTrigger value="stations" className="flex items-center gap-2">
   289→                  <Music className="h-3.5 w-3.5" />
   290→                  Emisoras ({loadedStations.length})
   291→                </TabsTrigger>
   292→                <TabsTrigger value="stats" className="flex items-center gap-2">
   293→                  <BarChart3 className="h-3.5 w-3.5" />
   294→                  Estadísticas
   295→                </TabsTrigger>
   296→              </TabsList>
   297→
   298→              <TabsContent value="stations" className="space-y-4 mt-0">
   299→                <div className="flex items-center justify-between">
   300→                  <h3 className="text-sm font-semibold uppercase tracking-wider text-muted-foreground">
   301→                    Gestiona tus emisoras
   302→                  </h3>
   303→                  <CreateStationDialog onCreate={handleCreateStation} />
   304→                </div>
   305→
   306→                {loadedStations.length > 1 && (
   307→                  <p className="text-[11px] text-muted-foreground bg-secondary/40 rounded-md px-3 py-1.5">
   308→                    💡 Usa los botones <strong>▲</strong> y <strong>▼</strong> junto a cada emisora para cambiar el orden en que aparecen en la página principal.
   309→                  </p>
   310→                )}
   311→
   312→                {loadedStations.length === 0 ? (
   313→                  <p className="rounded-lg border border-dashed border-border/60 p-6 text-center text-sm text-muted-foreground">
   314→                    Cargando emisoras…
   315→                  </p>
   316→                ) : (
   317→                  <div className="space-y-4">
   318→                    {[...loadedStations]
   319→                      .sort((a, b) => a.order - b.order)
   320→                      .map((station, sortedIndex) => (
   321→                      <StationAdminCard
   322→                        key={station.id}
   323→                        station={station}
   324→                        stationIndex={sortedIndex}
   325→                        totalStations={loadedStations.length}
   326→                        onMoveStation={(dir) => handleMoveStation(sortedIndex, dir)}
   327→                        onEdit={() => setEditingStation(station)}
   328→                        onDelete={() => handleDeleteStation(station.id, station.name)}
   329→                        onAddSong={(data) => handleAddSong(station.id, data)}
   330→                        onMoveSong={(index, dir) => handleMoveSong(station.id, index, dir)}
   331→                        onDeleteSong={(songId) => handleDeleteSong(songId, station.id)}
   332→                        onUpdateSong={(songId, data) => handleUpdateSong(songId, station.id, data)}
   333→                        onUpdateStation={(data) => handleUpdateStation(station.id, data)}
   334→                      />
   335→                    ))}
   336→                  </div>
   337→                )}
   338→              </TabsContent>
   339→
   340→              <TabsContent value="stats" className="mt-0">
   341→                <AdminStats />
   342→              </TabsContent>
   343→            </Tabs>
   344→          </div>
   345→        </ScrollArea>
   346→      </SheetContent>
   347→    </Sheet>
   348→  )
   349→}

   351→/**
   352→ * StationAdminCard
   353→ * Card for managing a single station: edit metadata, list songs, add song.
   354→ */
   355→function StationAdminCard({
   356→  station,
   357→  stationIndex,
   358→  totalStations,
   359→  onMoveStation,
   360→  onEdit,
   361→  onDelete,
   362→  onAddSong,
   363→  onMoveSong,
   364→  onDeleteSong,
   365→  onUpdateSong,
   366→  onUpdateStation,
   367→}: {
   368→  station: StationWithSongs
   369→  stationIndex: number
   370→  totalStations: number
   371→  onMoveStation: (dir: -1 | 1) => void
   372→  onEdit: () => void
   373→  onDelete: () => void
   374→  onAddSong: (data: { title: string; artist?: string; audioUrl: string; duration?: number; coverUrl?: string }) => void
   375→  onMoveSong: (index: number, dir: -1 | 1) => void
   376→  onDeleteSong: (songId: string) => void
   377→  onUpdateSong: (songId: string, data: Partial<Song>) => void
   378→  onUpdateStation: (data: Partial<Station>) => void
   379→}) {
   380→  const [expanded, setExpanded] = useState(true)
   381→  const [showAddSong, setShowAddSong] = useState(false)
   382→  const [editingSongId, setEditingSongId] = useState<string | null>(null)
   383→  const [editingStationOpen, setEditingStationOpen] = useState(false)
   384→
   385→  const accent = station.color
   386→  const songs = [...station.songs].sort((a, b) => a.order - b.order)
   387→
   388→  return (
   389→    <div
   390→      className="overflow-hidden rounded-xl border"
   391→      style={{ borderColor: `color-mix(in oklch, ${accent} 25%, transparent)` }}
   392→    >
   393→      {/* Station header */}
   394→      <div
   395→        className="flex items-center gap-3 p-4"
   396→        style={{
   397→          background: `linear-gradient(135deg, color-mix(in oklch, ${accent} 18%, transparent) 0%, transparent 100%)`,
   398→        }}
   399→      >
   400→        {/* Position badge + reorder controls */}
   401→        <div className="flex shrink-0 flex-col items-center gap-0.5">
   402→          <span
   403→            className="flex h-6 w-6 items-center justify-center rounded-full text-[10px] font-bold tabular-nums text-white"
   404→            style={{ background: accent }}
   405→            title={`Posición ${stationIndex + 1}`}
   406→          >
   407→            {stationIndex + 1}
   408→          </span>
   409→          <div className="flex flex-col">
   410→            <Button
   411→              variant="ghost"
   412→              size="icon"
   413→              className="h-5 w-5"
   414→              onClick={() => onMoveStation(-1)}
   415→              disabled={stationIndex === 0}
   416→              aria-label="Subir emisora"
   417→              title="Mover arriba"
   418→            >
   419→              <ArrowUp className="h-3 w-3" />
   420→            </Button>
   421→            <Button
   422→              variant="ghost"
   423→              size="icon"
   424→              className="h-5 w-5"
   425→              onClick={() => onMoveStation(1)}
   426→              disabled={stationIndex === totalStations - 1}
   427→              aria-label="Bajar emisora"
   428→              title="Mover abajo"
   429→            >
   430→              <ArrowDown className="h-3 w-3" />
   431→            </Button>
   432→          </div>
   433→        </div>
   434→
   435→        <div
   436→          className="h-10 w-10 shrink-0 rounded-full"
   437→          style={{
   438→            background: `radial-gradient(circle at 30% 30%, ${accent}, color-mix(in oklch, ${accent} 40%, black))`,
   439→            boxShadow: `0 0 16px -2px ${accent}`,
   440→          }}
   441→        />
   442→        <div className="min-w-0 flex-1">
   443→          <h4 className="truncate font-bold text-foreground">{station.name}</h4>
   444→          <p className="text-xs text-muted-foreground">
   445→            {station.language} · {songs.length} {songs.length === 1 ? 'canción' : 'canciones'}
   446→          </p>
   447→        </div>
   448→        <div className="flex shrink-0 items-center gap-1">
   449→          <Button
   450→            variant="ghost"
   451→            size="icon"
   452→            className="h-8 w-8"
   453→            onClick={() => setEditingStationOpen(true)}
   454→            aria-label="Editar emisora"
   455→          >
   456→            <Pencil className="h-3.5 w-3.5" />
   457→          </Button>
   458→          <Button
   459→            variant="ghost"
   460→            size="icon"
   461→            className="h-8 w-8 hover:text-destructive"
   462→            onClick={onDelete}
   463→            aria-label="Eliminar emisora"
   464→          >
   465→            <Trash2 className="h-3.5 w-3.5" />
   466→          </Button>
   467→          <Button
   468→            variant="ghost"
   469→            size="icon"
   470→            className="h-8 w-8"
   471→            onClick={() => setExpanded((e) => !e)}
   472→            aria-label={expanded ? 'Contrair' : 'Expandir'}
   473→          >
   474→            <X
   475→              className={cn('h-3.5 w-3.5 transition-transform', expanded ? 'rotate-45' : '')}
   476→            />
   477→          </Button>
   478→        </div>
   479→      </div>
   480→
   481→      {/* Songs list */}
   482→      {expanded && (
   483→        <div className="border-t border-border/60 p-3">
   484→          {songs.length === 0 ? (
   485→            <p className="py-4 text-center text-xs text-muted-foreground">
   486→              No hay canciones. Agrega la primera.
   487→            </p>
   488→          ) : (
   489→            <ul className="space-y-1">
   490→              {songs.map((song, index) => (
   491→                <li
   492→                  key={song.id}
   493→                  className="group flex items-center gap-2 rounded-md px-2 py-1.5 hover:bg-secondary"
   494→                >
   495→                  <GripVertical className="h-3.5 w-3.5 text-muted-foreground/40" />
   496→                  <span className="w-6 text-center text-[10px] tabular-nums text-muted-foreground">
   497→                    {index + 1}
   498→                  </span>
   499→                  <div className="min-w-0 flex-1">
   500→                    <p className="truncate text-xs font-medium text-foreground">
   501→                      {song.title}
   502→                    </p>
   503→                    <p className="truncate text-[10px] text-muted-foreground">
   504→                      {song.artist} · {formatTime(song.duration)}
   505→                    </p>
   506→                  </div>
   507→                  <div className="flex items-center gap-0.5 opacity-0 transition-opacity group-hover:opacity-100">
   508→                    <Button
   509→                      variant="ghost"
   510→                      size="icon"
   511→                      className="h-6 w-6"
   512→                      onClick={() => onMoveSong(index, -1)}
   513→                      disabled={index === 0}
   514→                      aria-label="Subir"
   515→                    >
   516→                      <ArrowUp className="h-3 w-3" />
   517→                    </Button>
   518→                    <Button
   519→                      variant="ghost"
   520→                      size="icon"
   521→                      className="h-6 w-6"
   522→                      onClick={() => onMoveSong(index, 1)}
   523→                      disabled={index === songs.length - 1}
   524→                      aria-label="Bajar"
   525→                    >
   526→                      <ArrowDown className="h-3 w-3" />
   527→                    </Button>
   528→                    <Button
   529→                      variant="ghost"
   530→                      size="icon"
   531→                      className="h-6 w-6"
   532→                      onClick={() => setEditingSongId(song.id)}
   533→                      aria-label="Editar"
   534→                    >
   535→                      <Pencil className="h-3 w-3" />
   536→                    </Button>
   537→                    <a
   538→                      href={song.audioUrl}
   539→                      target="_blank"
   540→                      rel="noopener noreferrer"
   541→                      className="inline-flex h-6 w-6 items-center justify-center rounded hover:bg-secondary"
   542→                      aria-label="Abrir audio"
   543→                    >
   544→                      <ExternalLink className="h-3 w-3" />
   545→                    </a>
   546→                    <Button
   547→                      variant="ghost"
   548→                      size="icon"
   549→                      className="h-6 w-6 hover:text-destructive"
   550→                      onClick={() => onDeleteSong(song.id)}
   551→                      aria-label="Eliminar"
   552→                    >
   553→                      <Trash2 className="h-3 w-3" />
   554→                    </Button>
   555→                  </div>
   556→                </li>
   557→              ))}
   558→            </ul>
   559→          )}
   560→
   561→          <Button
   562→            variant="outline"
   563→            size="sm"
   564→            className="mt-3 w-full border-dashed"
   565→            onClick={() => setShowAddSong(true)}
   566→          >
   567→            <Plus className="h-3.5 w-3.5" />
   568→            Agregar canción
   569→          </Button>
   570→        </div>
   571→      )}
   572→
   573→      {showAddSong && (
   574→        <AddSongDialog
   575→          open={showAddSong}
   576→          onOpenChange={setShowAddSong}
   577→          onSubmit={(data) => {
   578→            onAddSong(data)
   579→            setShowAddSong(false)
   580→          }}
   581→          accent={accent}
   582→          stationId={station.id}
   583→        />
   584→      )}
   585→
   586→      {editingSongId && (
   587→        <EditSongDialog
   588→          key={editingSongId}
   589→          song={songs.find((s) => s.id === editingSongId)!}
   590→          open={!!editingSongId}
   591→          onOpenChange={(o) => !o && setEditingSongId(null)}
   592→          onSubmit={(data) => {
   593→            onUpdateSong(editingSongId!, data)
   594→            setEditingSongId(null)
   595→          }}
   596→          accent={accent}
   597→          stationId={station.id}
   598→        />
   599→      )}
   600→
   601→      <EditStationDialog
   602→        key={station.id}
   603→        station={station}
   604→        open={editingStationOpen}
   605→        onOpenChange={setEditingStationOpen}
   606→        onSubmit={(data) => {
   607→          onUpdateStation(data)
   608→          setEditingStationOpen(false)
   609→        }}
   610→      />
   611→    </div>
   612→  )
   613→}
